import React, { useState, useEffect, useCallback } from "react";
import "./Profitability.scss";
import BarChart from "../../commons/bar-chart/BarChart";
import Tooltip from "../../commons/tooltip/Tooltip";

const SectionProfitability = ({ afps, funds, periods, chart, information, dateRange }) => {
    const [openTooltipProfitability, setOpenTooltipProfitability] = useState(false);
    const [openTooltipLegal, setOpenTooltipLegal] = useState(false);
    const [currentFund, setCurrentFund] = useState("");
    const [currentPeriod, setCurrentPeriod] = useState("");
    const [currentDateRange, setCurrentDateRange] = useState("");

    useEffect(() => {
        if (!funds || !periods || !dateRange) return;

        const activeFund = funds.list.find((item) => item.active === true);
        const currentFund = funds.list.length > 0 ? activeFund?.id : "";

        const activePeriod = periods.list.find((item) => item.active === true);
        const currentPeriod = periods.list.length > 0 ? activePeriod?.period_id : "";

        const currentDateRange = dateRange.length > 0 ? dateRange[0].text : "";
        setCurrentFund(currentFund);
        setCurrentPeriod(currentPeriod);
        setCurrentDateRange(currentDateRange);
    }, [funds, periods, dateRange]);

    const toggleTooltipProfitability = useCallback((event) => {
        event.preventDefault();
        setOpenTooltipProfitability((prev) => !prev);
    }, []);

    const toggleTooltipLegal = useCallback((event) => {
        event.preventDefault();
        setOpenTooltipLegal((prev) => !prev);
    }, []);

    const onChangeValue = (name, value) => {
        let range;
        if (name === "currentFund") {
            const found = dateRange.filter((item) => item.fund_id === value);
            range = found.find((item) => item.period_id === currentPeriod);
        } else {
            const period = dateRange.filter((item) => item.period_id === value);
            range = period.find((item) => item.fund_id === currentFund);
        }
        if (name === "currentFund") {
            setCurrentFund(value);
        } else {
            setCurrentPeriod(value);
        }
        setCurrentDateRange(range?.text || "");
    };

    const getBars = () => {
        const bars = [];
        chart.bar.forEach((bar) => {
            if (bar.fundId === currentFund && bar.periodId === currentPeriod) {
                const afp = afps.find((afp) => afp.afp_id === bar.afpId);
                if (afp) {
                    bars.push({
                        name: afp.name,
                        value: Number(bar.value),
                        label: bar.value,
                        color: afp.color,
                    });
                }
            }
        });
        return bars;
    };

    const toolTipDataProfitability = {
        title: information.reference.popPup.title,
        question: information.reference.popPup.question,
        description: information.reference.popPup.description,
    };

    const toolTipDataLegal = {
        title: information.btnInformation.reference.popPup.title,
        description: information.btnInformation.reference.popPup.description,
    };

    const tagA = () => {
        const reference = information.reference;
        return (
            <a href="#!" onClick={toggleTooltipProfitability}>
                {reference.name}
            </a>
        );
    };

    const tagP = () => {
        let subTitle = information.subTitle;
        subTitle = subTitle.split("#");
        return (
            <p className={`hbi-text-profitability`}>
                {subTitle[0]} {tagA()} {subTitle[2]} / <strong>{currentDateRange}</strong>
            </p>
        );
    };

    const tagBtnInformation = () => {
        let title = information.btnInformation.title;
        title = title.split("#");
        return (
            <a
                href={"#!"}
                className={`btn btn-legal-information`}
                onClick={toggleTooltipLegal}
            >
                {title[0]} <strong>{information.btnInformation.reference.name}</strong>
            </a>
        );
    };

    const tagSplitStrong = () => {
        let description = toolTipDataLegal.description;
        let subString = description.split("#");
        let cambioAnios = subString[2].split("AFP Abril 2019 / Abril 2014");
        return (
            <div>
                <strong>{subString[1]}</strong> {cambioAnios[0]} {currentDateRange} {cambioAnios[1]}
                <strong>{subString[3]}</strong>
            </div>
        );
    };

    return (
        <section id={"resultados-de-rentabilidad"} className={`general-profitability grafico1`}>
            <div className={`container hbi-section-profitability`}>
                <div className={`text-center hbi-reference-topltip hbi-reference-topltip--custom`}>
                    <h2 className={`title-profitability`}>{information.title}</h2>
                    {tagP()}
                    {openTooltipProfitability && <Tooltip {...toolTipDataProfitability} />}
                </div>
                <div className={`hbi-content-tabs-fondo`}>
                    <ul className={`hbi-tabs-fondo`}>
                        {funds.list.map((fund) => (
                            <li
                                key={fund.id}
                                className={`hbi-item-fondo ${currentFund === fund.id ? "hbi-item-fondo--active" : ""}`}
                                onClick={() => onChangeValue("currentFund", fund.id)}
                            >
                                {fund.name}
                            </li>
                        ))}
                    </ul>
                </div>
                <div className={`hbi-cardbox`}>
                    <BarChart bars={getBars()} />
                </div>
                <div className={`hbi-content-years`}>
                    <ul className={`hbi-years`} style={{ overflow: "auto" }}>
                        {periods.list.map((period) => (
                            <li key={period.period_id} className={`hbi-item-year`}>
                                <button
                                    className={`btn btn-year ${currentPeriod === period.period_id ? "btn-year--active" : ""}`}
                                    onClick={() => onChangeValue("currentPeriod", period.period_id)}
                                >
                                    {period.name}
                                </button>
                            </li>
                        ))}
                    </ul>
                </div>
            </div>
            <div className={`hbi-content-legal hbi-reference-topltip d-none`}>
                {tagBtnInformation()}
                {openTooltipLegal && <Tooltip {...toolTipDataLegal} />}
            </div>
            <div className={`container`}>
                <div className={`tooltip-reference`}>
                    <div>{tagSplitStrong()}</div>
                </div>
            </div>
        </section>
    );
};

export default SectionProfitability;

  const hasResponse = error.response && typeof error.response.data === 'object';
  if (hasResponse) {
    const errorData = error.response.data.error;
    return { ...ERROR_PAYLOAD, data: errorData, message: errorData.message };
  }
  return ERROR_PAYLOAD;
};

const sendGet = async (uri, data, config) => {
  try {
    const requestConfig = { ...config, params: { ...data } };

    // const payload = await handleResponse(response);
    const payload = response.data;
    console.log(payload);
    return payload;
  } catch (error) {
    return handleError(uri, error);

};
